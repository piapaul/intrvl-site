/**
 * intrvl friends leaderboard — Google Apps Script web app (version 2).
 *
 * Paste this whole file over the old Code.gs in your Sheet's Apps Script
 * project, then redeploy as a NEW VERSION (see docs/LEADERBOARD-SETUP.md).
 *
 * What changed in version 6 (sync, league manager, more games):
 *   - Tabs "sync" and "stats" are created for you; "league_members" gains two columns (share, managerName).
 *   - Sync: a player can save their progress online and restore it on another phone / the website with their
 *     friend code plus a recovery code. The recovery code never reaches the server, only a hash of it.
 *   - League manager: a member can choose to share a small progress summary with the league owner, shown under
 *     a First name + initial. Sharing can be turned off at any time, which also deletes the summary.
 *   - deleteMyData: removes everything stored about a player (the owner must delete their leagues first).
 *   - Score limits for Interval Reader (+Easy) and Chord Quality.
 *
 * What changed in version 5 (Friends League):
 *   - Two new tabs, "leagues" and "league_members", are created for you.
 *   - A league is a named group with an owner and an invite link. Everyone's daily scores for the week are turned
 *     into league points (1 for every daily game finished, +2 for the best score of the day, +1 for second best).
 *   - Only the owner can rename, remove members, reset the invite link or delete the league.
 *   - Admin actions are checked against the device that registered the player code, not just the code.
 *   - Score limits added for the new games (Easy sprints, Key Signature Sprint, Rhythm Tap).
 *
 * What changed in version 4 (hardening - see docs in the README, "The leaderboard secret"):
 *   - The shared secret is a speed bump, not real security: it is visible in the page source of a public
 *     website, so anyone can read it. What actually protects the leaderboard is below.
 *   - Every write is rate limited (per device and across the whole app), so one person cannot flood the Sheet.
 *   - Device ids must look like the ids the pages generate; names, scores, streaks and friend counts are capped.
 *   - Optional second secret (SHARED_SECRET_NEXT) so the secret can be rotated without breaking the live site.
 *
 * What changed from version 1:
 *   - The three sheet tabs (scores / players / friends) are created for you,
 *     with the right headers and with date/code columns formatted as TEXT, so
 *     Google Sheets can no longer turn "2026-10-06" into a real date and break
 *     the comparison (that was the main reason the leaderboard "didn't work").
 *   - Dates that were already converted by Sheets are repaired when read.
 *   - Friends are stored on the server and are MUTUAL: when A adds B, B also
 *     sees A. Removing a friend removes both directions.
 *   - The page sends its own local date, so a player in another time zone
 *     still sees "today" correctly.
 *   - A duplicate score submission is treated as success.
 *   - ?action=ping tells you the backend is alive and which version it is.
 *
 * Script Property needed (Project Settings → Script properties):
 *   SHARED_SECRET = the same text as `secret` in src/config.js
 *
 * All responses are HTTP 200 (Apps Script can't set status codes); callers
 * read the `ok` field in the JSON.
 */

// ---------------------------------------------------------------- entry points

function doGet(e) {
  try {
    var p = (e && e.parameter) || {};
    switch (p.action) {
      case 'ping': return ping_();
      case 'resolveCode': return resolveCode_(p);
      case 'friends': return friendsBoard_(p);
      case 'board': return myBoard_(p);
      case 'friendsOf': return friendsOf_(p);
      case 'myLeagues': return myLeagues_(p);
      case 'league': return leagueView_(p);
      case 'invitePreview': return invitePreview_(p);
      default: return json_({ ok: false, error: 'unknown action' });
    }
  } catch (err) {
    return json_({ ok: false, error: 'server error: ' + err });
  }
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  try {
    var body;
    try { body = JSON.parse(e.postData.contents); } catch (x) { return json_({ ok: false, error: 'invalid JSON body' }); }
    var props = PropertiesService.getScriptProperties();
    var secret = props.getProperty('SHARED_SECRET'), nextSecret = props.getProperty('SHARED_SECRET_NEXT');
    var okSecret = !!secret && (body.secret === secret || (!!nextSecret && body.secret === nextSecret));
    if (!okSecret) return json_({ ok: false, error: 'unauthorized' });
    if (!validDevice_(body.deviceId) && body.action !== 'addFriend' && body.action !== 'removeFriend' && body.action !== 'syncRestore') return json_({ ok: false, error: 'invalid deviceId' });

    lock.waitLock(15000);
    if (tooManyWrites_(body.deviceId || body.code)) return json_({ ok: false, error: 'slow down — too many requests' });
    switch (body.action) {
      case 'registerPlayer': return registerPlayer_(body);
      case 'addFriend': return addFriend_(body);
      case 'removeFriend': return removeFriend_(body);
      case 'createLeague': return createLeague_(body);
      case 'joinLeague': return joinLeague_(body);
      case 'leaveLeague': return leaveLeague_(body);
      case 'removeMember': return removeMember_(body);
      case 'renameLeague': return renameLeague_(body);
      case 'resetInvite': return resetInvite_(body);
      case 'deleteLeague': return deleteLeague_(body);
      case 'setMemberPrefs': return setMemberPrefs_(body);
      case 'statsPush': return statsPush_(body);
      case 'leagueManage': return leagueManage_(body);
      case 'syncEnable': return syncEnable_(body);
      case 'syncPush': return syncPush_(body);
      case 'syncPull': return syncPull_(body);
      case 'syncRestore': return syncRestore_(body);
      case 'deleteMyData': return deleteMyData_(body);
      default: return submitScore_(body);       // no action = a score submission
    }
  } catch (err) {
    return json_({ ok: false, error: 'server error: ' + err });
  } finally {
    try { lock.releaseLock(); } catch (x2) { /* not held */ }
  }
}

// You can also run this by hand once (Run ▶ setup) to create the tabs.
function setup() {
  Object.keys(HEADERS).forEach(function (n) { sheet_(n); });
  PropertiesService.getScriptProperties().setProperty('SETUP_V', String(VERSION));
  return 'ok';
}

// ---------------------------------------------------------------- handlers

function ping_() {
  setup();
  return json_({ ok: true, version: VERSION, serverDate: today_(), timezone: Session.getScriptTimeZone() });
}

function registerPlayer_(b) {
  var code = cleanCode_(b.code);
  if (!code) return json_({ ok: false, error: 'invalid code' });
  if (!validDevice_(b.deviceId)) return json_({ ok: false, error: 'missing deviceId' });

  var sh = sheet_(PLAYERS), vals = sh.getDataRange().getValues();
  var name = cleanName_(b.displayName), avatar = cleanAvatar_(b.avatar), now = new Date().toISOString();
  for (var i = 1; i < vals.length; i++) {
    if (String(vals[i][0]).toUpperCase() === code) {
      if (String(vals[i][1]) !== b.deviceId) return json_({ ok: false, error: 'code taken' });
      sh.getRange(i + 1, 2, 1, 4).setValues([[b.deviceId, name, avatar, now]]);
      return json_({ ok: true });
    }
  }
  if (vals.length > LIMITS.maxPlayers) return json_({ ok: false, error: 'the leaderboard is full right now' });
  appendText_(sh, 'players', [code, b.deviceId, name, avatar, now]);
  return json_({ ok: true });
}

function submitScore_(b) {
  if (!SCORE_BOUNDS.hasOwnProperty(b.gameId)) return json_({ ok: false, error: 'unknown gameId' });
  if (!validDevice_(b.deviceId)) return json_({ ok: false, error: 'missing deviceId' });
  var date = String(b.dailySeedDate || '');
  if (!/^\d{4}-\d{2}-\d{2}$/.test(date)) return json_({ ok: false, error: 'missing/invalid dailySeedDate' });
  if (Math.abs(daysBetween_(date, today_())) > 2) return json_({ ok: false, error: 'date out of range' });
  var score = Number(b.score), bounds = SCORE_BOUNDS[b.gameId];
  if (isNaN(score) || score < bounds[0] || score > bounds[1]) return json_({ ok: false, error: 'score out of range for this game' });

  var sh = sheet_(SCORES), vals = sh.getDataRange().getValues();
  for (var i = 1; i < vals.length; i++) {
    if (String(vals[i][1]) === b.gameId && String(vals[i][2]) === b.deviceId && normDate_(vals[i][4]) === date) {
      return json_({ ok: true, duplicate: true });          // first score of the day stands
    }
  }
  appendText_(sh, 'scores', [new Date().toISOString(), b.gameId, b.deviceId, cleanName_(b.displayName), date, score, Math.max(0, Math.min(LIMITS.maxStreak, Number(b.streak) || 0))]);
  return json_({ ok: true });
}

function resolveCode_(p) {
  var code = cleanCode_(p.code);
  var player = code && playersByCode_()[code];
  if (!player) return json_({ ok: false, error: 'code not found' });
  return json_({ ok: true, code: code, displayName: player.displayName, avatar: player.avatar });
}

function addFriend_(b) {
  var a = cleanCode_(b.code), f = cleanCode_(b.friendCode);
  if (!a || !f) return json_({ ok: false, error: 'invalid code' });
  if (a === f) return json_({ ok: false, error: "that's your own code" });
  var players = playersByCode_();
  if (!players[a]) return json_({ ok: false, error: 'your code is not registered yet — try again' });
  if (!players[f]) return json_({ ok: false, error: 'code not found' });

  var sh = sheet_(FRIENDS), edges = sh.getDataRange().getValues();
  if (!edgeExists_(edges, a, f)) {
    if (countFriends_(edges, a) >= LIMITS.maxFriendsPerPlayer || countFriends_(edges, f) >= LIMITS.maxFriendsPerPlayer) return json_({ ok: false, error: 'friend limit reached' });
    appendText_(sh, 'friends', [a, f, new Date().toISOString()]);
  }
  return json_({ ok: true, friend: { code: f, displayName: players[f].displayName, avatar: players[f].avatar } });
}

function removeFriend_(b) {
  var a = cleanCode_(b.code), f = cleanCode_(b.friendCode);
  if (!a || !f) return json_({ ok: false, error: 'invalid code' });
  var sh = sheet_(FRIENDS), vals = sh.getDataRange().getValues();
  for (var i = vals.length - 1; i >= 1; i--) {            // bottom-up so row numbers stay valid
    var x = String(vals[i][0]).toUpperCase(), y = String(vals[i][1]).toUpperCase();
    if ((x === a && y === f) || (x === f && y === a)) sh.deleteRow(i + 1);
  }
  return json_({ ok: true });
}

function friendsOf_(p) {
  var code = cleanCode_(p.code);
  if (!code) return json_({ ok: false, error: 'invalid code' });
  var players = playersByCode_(), vals = sheet_(FRIENDS).getDataRange().getValues(), out = [], seen = {};
  for (var i = 1; i < vals.length; i++) {
    var x = String(vals[i][0]).toUpperCase(), y = String(vals[i][1]).toUpperCase(), other = x === code ? y : y === code ? x : null;
    if (other && !seen[other]) {
      seen[other] = true;
      var pl = players[other];
      out.push({ code: other, displayName: pl ? pl.displayName : other, avatar: pl ? pl.avatar : DEFAULT_AVATAR });
    }
  }
  return json_({ ok: true, friends: out });
}

// codes=A,B,C  date=YYYY-MM-DD  [gameId=x]  -> each player's scores for that date (all games) + their score for gameId
function friendsBoard_(p) {
  var codes = String(p.codes || '').split(',').map(cleanCode_).filter(Boolean).slice(0, 60);
  return json_(buildBoard_(codes, p.date, p.gameId, playersByCode_()));
}

// ONE call for the whole Friends screen: code=ME [codes=local,friends] -> board of me + every friend
// (friends stored on the server + any the phone sends), so the page needs a single round trip.
function myBoard_(p) {
  var me = cleanCode_(p.code);
  if (!me) return json_({ ok: false, error: 'invalid code' });
  var players = playersByCode_(), seen = {}, codes = [me];
  seen[me] = true;
  var vals = sheet_(FRIENDS).getDataRange().getValues();
  for (var i = 1; i < vals.length; i++) {
    var x = String(vals[i][0]).toUpperCase(), y = String(vals[i][1]).toUpperCase(), other = x === me ? y : y === me ? x : null;
    if (other && !seen[other]) { seen[other] = true; codes.push(other); }
  }
  String(p.codes || '').split(',').map(cleanCode_).filter(Boolean).forEach(function (c) { if (!seen[c] && players[c]) { seen[c] = true; codes.push(c); } });
  var out = buildBoard_(codes.slice(0, 60), p.date, p.gameId, players);
  out.friendCodes = codes.slice(1);
  return json_(out);
}

function buildBoard_(codes, dateParam, gameParam, players) {
  var date = /^\d{4}-\d{2}-\d{2}$/.test(dateParam || '') ? dateParam : today_();
  var gameId = SCORE_BOUNDS.hasOwnProperty(gameParam) ? gameParam : null, byDevice = {};

  var vals = sheet_(SCORES).getDataRange().getValues();
  for (var i = 1; i < vals.length; i++) {
    if (normDate_(vals[i][4]) !== date) continue;
    var dev = String(vals[i][2]), g = String(vals[i][1]), sc = Number(vals[i][5]);
    byDevice[dev] = byDevice[dev] || {};
    if (byDevice[dev][g] == null || sc > byDevice[dev][g]) byDevice[dev][g] = sc;
  }

  var list = codes.map(function (code) {
    var pl = players[code];
    if (!pl) return { code: code, displayName: code, avatar: null, scores: {}, playedCount: 0, hasPlayed: false, score: null };
    var scores = byDevice[pl.deviceId] || {};
    var has = gameId ? scores[gameId] != null : Object.keys(scores).length > 0;
    return { code: code, displayName: pl.displayName, avatar: pl.avatar, scores: scores, playedCount: Object.keys(scores).length, hasPlayed: has, score: gameId && has ? scores[gameId] : null };
  });
  list.sort(function (a, b) {
    if (a.hasPlayed !== b.hasPlayed) return a.hasPlayed ? -1 : 1;
    if (gameId && a.hasPlayed) return b.score - a.score;
    return b.playedCount - a.playedCount;
  });
  return { ok: true, date: date, gameId: gameId, friends: list };
}


// ---------------------------------------------------------------- friends leagues

var ID_ALPHABET = 'ABCDEFGHJKMNPQRSTUVWXYZ23456789';
function randomId_(len) {
  var s = '';
  for (var i = 0; i < len; i++) s += ID_ALPHABET.charAt(Math.floor(Math.random() * ID_ALPHABET.length));
  return s;
}
function cleanLeagueName_(n) {
  var t = typeof n === 'string' ? n.replace(/[<>]/g, '').replace(/\s+/g, ' ').trim().slice(0, 30) : '';
  return t;
}

// The caller must be a registered player AND prove it with the device id that registered the code.
function authPlayer_(code, deviceId) {
  var c = cleanCode_(code);
  if (!c) return { error: 'invalid code' };
  if (!validDevice_(deviceId)) return { error: 'missing deviceId' };
  var pl = playersByCode_()[c];
  if (!pl) return { error: 'your code is not registered yet — try again' };
  if (pl.deviceId !== String(deviceId)) return { error: 'not allowed' };
  return { code: c, player: pl };
}

function leagueRows_() { return sheet_(LEAGUES).getDataRange().getValues(); }
function memberRows_() { return sheet_(MEMBERS).getDataRange().getValues(); }
function findLeague_(rows, leagueId) {
  for (var i = 1; i < rows.length; i++) if (String(rows[i][0]) === String(leagueId)) return { row: i + 1, id: String(rows[i][0]), name: String(rows[i][1]), ownerCode: String(rows[i][2]).toUpperCase(), invite: String(rows[i][4]) };
  return null;
}
function membersOf_(mrows, leagueId) {
  var out = [];
  for (var i = 1; i < mrows.length; i++) if (String(mrows[i][0]) === String(leagueId)) out.push({ code: String(mrows[i][1]).toUpperCase(), role: String(mrows[i][2]), joinedAt: String(mrows[i][3]), share: String(mrows[i][4]) !== '0', managerName: String(mrows[i][5] || ''), row: i + 1 });
  return out;
}
function flag_(v) { return (v === false || v === 0 || v === '0' || v === 'false') ? '0' : '1'; }
function cleanManagerName_(n) {
  return typeof n === 'string' ? n.replace(/[^A-Za-zÀ-ÖØ-öø-ÿ'’\- ]/g, '').replace(/\s+/g, ' ').trim().slice(0, 22) : '';
}
function leaguesOf_(mrows, code) {
  var out = [];
  for (var i = 1; i < mrows.length; i++) if (String(mrows[i][1]).toUpperCase() === code) out.push({ leagueId: String(mrows[i][0]), role: String(mrows[i][2]) });
  return out;
}
function isMember_(mrows, leagueId, code) {
  for (var i = 1; i < mrows.length; i++) if (String(mrows[i][0]) === String(leagueId) && String(mrows[i][1]).toUpperCase() === code) return true;
  return false;
}
function deleteMemberRows_(leagueId, code) {            // code = null removes everyone in the league
  var sh = sheet_(MEMBERS), vals = sh.getDataRange().getValues();
  for (var i = vals.length - 1; i >= 1; i--) {
    if (String(vals[i][0]) === String(leagueId) && (code === null || String(vals[i][1]).toUpperCase() === code)) sh.deleteRow(i + 1);
  }
}

function createLeague_(b) {
  var a = authPlayer_(b.code, b.deviceId);
  if (a.error) return json_({ ok: false, error: a.error });
  var name = cleanLeagueName_(b.name);
  if (!name) return json_({ ok: false, error: 'give the league a name' });
  var lrows = leagueRows_(), mrows = memberRows_(), owned = 0, joined = 0, i;
  for (i = 1; i < lrows.length; i++) if (String(lrows[i][2]).toUpperCase() === a.code) owned++;
  joined = leaguesOf_(mrows, a.code).length;
  if (owned >= LIMITS.maxLeaguesOwned) return json_({ ok: false, error: 'you can run up to ' + LIMITS.maxLeaguesOwned + ' leagues' });
  if (joined >= LIMITS.maxLeaguesJoined) return json_({ ok: false, error: 'you are in the maximum number of leagues' });
  var id = randomId_(8), invite = randomId_(8), now = new Date().toISOString();
  appendText_(sheet_(LEAGUES), 'leagues', [id, name, a.code, now, invite]);
  appendText_(sheet_(MEMBERS), 'league_members', [id, a.code, 'owner', now, flag_(b.share), cleanManagerName_(b.managerName)]);
  return json_({ ok: true, league: { leagueId: id, name: name, inviteCode: invite } });
}

function joinLeague_(b) {
  var a = authPlayer_(b.code, b.deviceId);
  if (a.error) return json_({ ok: false, error: a.error });
  var invite = String(b.invite || '').trim().toUpperCase();
  if (!/^[A-Z0-9]{6,12}$/.test(invite)) return json_({ ok: false, error: 'invite not found' });
  var lrows = leagueRows_(), found = null, i;
  for (i = 1; i < lrows.length; i++) if (String(lrows[i][4]).toUpperCase() === invite) { found = findLeague_(lrows, lrows[i][0]); break; }
  if (!found) return json_({ ok: false, error: 'invite not found' });
  var mrows = memberRows_();
  if (isMember_(mrows, found.id, a.code)) return json_({ ok: true, alreadyMember: true, league: { leagueId: found.id, name: found.name } });
  if (membersOf_(mrows, found.id).length >= LIMITS.maxLeagueMembers) return json_({ ok: false, error: 'that league is full' });
  if (leaguesOf_(mrows, a.code).length >= LIMITS.maxLeaguesJoined) return json_({ ok: false, error: 'you are in the maximum number of leagues' });
  appendText_(sheet_(MEMBERS), 'league_members', [found.id, a.code, 'member', new Date().toISOString(), flag_(b.share), cleanManagerName_(b.managerName)]);
  return json_({ ok: true, league: { leagueId: found.id, name: found.name } });
}

function leaveLeague_(b) {
  var a = authPlayer_(b.code, b.deviceId);
  if (a.error) return json_({ ok: false, error: a.error });
  var lg = findLeague_(leagueRows_(), b.leagueId);
  if (!lg) return json_({ ok: false, error: 'league not found' });
  if (lg.ownerCode === a.code) return json_({ ok: false, error: 'the owner can’t leave — delete the league instead' });
  deleteMemberRows_(lg.id, a.code);
  return json_({ ok: true });
}

function ownerOnly_(b) {
  var a = authPlayer_(b.code, b.deviceId);
  if (a.error) return { error: a.error };
  var lg = findLeague_(leagueRows_(), b.leagueId);
  if (!lg) return { error: 'league not found' };
  if (lg.ownerCode !== a.code) return { error: 'only the league owner can do that' };
  return { a: a, lg: lg };
}

function removeMember_(b) {
  var o = ownerOnly_(b);
  if (o.error) return json_({ ok: false, error: o.error });
  var who = cleanCode_(b.memberCode);
  if (!who) return json_({ ok: false, error: 'invalid code' });
  if (who === o.lg.ownerCode) return json_({ ok: false, error: 'you can’t remove the owner' });
  deleteMemberRows_(o.lg.id, who);
  return json_({ ok: true });
}

function renameLeague_(b) {
  var o = ownerOnly_(b);
  if (o.error) return json_({ ok: false, error: o.error });
  var name = cleanLeagueName_(b.name);
  if (!name) return json_({ ok: false, error: 'give the league a name' });
  var sh = sheet_(LEAGUES);
  sh.getRange(o.lg.row, 2).setValue(name);
  return json_({ ok: true, name: name });
}

function resetInvite_(b) {
  var o = ownerOnly_(b);
  if (o.error) return json_({ ok: false, error: o.error });
  var invite = randomId_(8);
  sheet_(LEAGUES).getRange(o.lg.row, 5).setValue(invite);
  return json_({ ok: true, inviteCode: invite });
}

function deleteLeague_(b) {
  var o = ownerOnly_(b);
  if (o.error) return json_({ ok: false, error: o.error });
  deleteMemberRows_(o.lg.id, null);
  sheet_(LEAGUES).deleteRow(o.lg.row);
  return json_({ ok: true });
}

function invitePreview_(p) {
  var invite = String(p.invite || '').trim().toUpperCase();
  if (!/^[A-Z0-9]{6,12}$/.test(invite)) return json_({ ok: false, error: 'invite not found' });
  var lrows = leagueRows_(), players = playersByCode_();
  for (var i = 1; i < lrows.length; i++) {
    if (String(lrows[i][4]).toUpperCase() === invite) {
      var lg = findLeague_(lrows, lrows[i][0]), owner = players[lg.ownerCode];
      return json_({ ok: true, name: lg.name, members: membersOf_(memberRows_(), lg.id).length, ownerName: owner ? owner.displayName : 'the owner' });
    }
  }
  return json_({ ok: false, error: 'invite not found' });
}

// Monday of the week containing a yyyy-MM-dd date
function mondayOf_(dateStr) {
  var d = new Date(dateStr + 'T00:00:00Z'), dow = (d.getUTCDay() + 6) % 7;
  d.setUTCDate(d.getUTCDate() - dow);
  return d.toISOString().slice(0, 10);
}
function addDays_(dateStr, n) {
  var d = new Date(dateStr + 'T00:00:00Z');
  d.setUTCDate(d.getUTCDate() + n);
  return d.toISOString().slice(0, 10);
}

// League points for the 7 days from startDate.  members: [{code, deviceId}]
// Per game per day: 1 point for playing, +2 for the best score, +1 for the next best (ties share).
function standings_(members, scoreRows, startDate) {
  var dates = [], d, i;
  for (i = 0; i < 7; i++) dates.push(addDays_(startDate, i));
  var devToCode = {}, table = {};
  members.forEach(function (m) { devToCode[m.deviceId] = m.code; table[m.code] = { code: m.code, points: 0, played: 0, days: [0, 0, 0, 0, 0, 0, 0] }; });
  var best = {};     // date|game|code -> best score
  for (i = 1; i < scoreRows.length; i++) {
    var dev = String(scoreRows[i][2]), code = devToCode[dev];
    if (!code) continue;
    var date = normDate_(scoreRows[i][4]), di = dates.indexOf(date);
    if (di < 0) continue;
    var key = date + '|' + String(scoreRows[i][1]) + '|' + code, sc = Number(scoreRows[i][5]);
    if (best[key] == null || sc > best[key]) best[key] = sc;
  }
  var groups = {};   // date|game -> [{code, score}]
  Object.keys(best).forEach(function (k) {
    var parts = k.split('|'), g = parts[0] + '|' + parts[1];
    (groups[g] = groups[g] || []).push({ code: parts[2], score: best[k], di: dates.indexOf(parts[0]) });
  });
  Object.keys(groups).forEach(function (g) {
    var list = groups[g], scores = [];
    list.forEach(function (x) { if (scores.indexOf(x.score) < 0) scores.push(x.score); });
    scores.sort(function (a, b) { return b - a; });
    list.forEach(function (x) {
      var pts = 1;
      if (x.score === scores[0]) pts += 2;
      else if (x.score === scores[1]) pts += 1;
      table[x.code].points += pts; table[x.code].played += 1; table[x.code].days[x.di] += pts;
    });
  });
  return table;
}

function leagueView_(p) {
  var code = cleanCode_(p.code);
  if (!code) return json_({ ok: false, error: 'invalid code' });
  var lrows = leagueRows_(), mrows = memberRows_(), lg = findLeague_(lrows, p.leagueId);
  if (!lg) return json_({ ok: false, error: 'league not found' });
  if (!isMember_(mrows, lg.id, code)) return json_({ ok: false, error: 'you are not in this league' });
  var players = playersByCode_(), roster = membersOf_(mrows, lg.id);
  var members = roster.map(function (m) { var pl = players[m.code]; return { code: m.code, deviceId: pl ? pl.deviceId : '', role: m.role, displayName: pl ? pl.displayName : m.code, avatar: pl ? pl.avatar : DEFAULT_AVATAR }; });
  var today = today_(), week = /^\d{4}-\d{2}-\d{2}$/.test(p.week || '') ? mondayOf_(p.week) : mondayOf_(today);
  var prevWeek = addDays_(week, -7), scoreRows = sheet_(SCORES).getDataRange().getValues();
  var cur = standings_(members, scoreRows, week), prev = standings_(members, scoreRows, prevWeek);
  var rows = members.map(function (m) {
    var s = cur[m.code];
    return { code: m.code, displayName: m.displayName, avatar: m.avatar, role: m.role, points: s.points, played: s.played, days: s.days };
  });
  rows.sort(function (a, b) { return b.points - a.points || b.played - a.played || (a.displayName < b.displayName ? -1 : 1); });
  var rank = 0, last = null, shown = 0;
  rows.forEach(function (r) { shown++; if (last === null || r.points !== last) rank = shown; r.rank = rank; last = r.points; });
  // last week's winner (only when somebody scored)
  var prevBest = null;
  members.forEach(function (m) { var s = prev[m.code]; if (s.points > 0 && (!prevBest || s.points > prevBest.points)) prevBest = { displayName: m.displayName, avatar: m.avatar, points: s.points }; });
  var isOwner = lg.ownerCode === code;
  return json_({
    ok: true,
    league: { leagueId: lg.id, name: lg.name, role: isOwner ? 'owner' : 'member', inviteCode: isOwner ? lg.invite : null },
    week: week, weekEnd: addDays_(week, 6), today: today, members: rows, lastWeekWinner: prevBest
  });
}

// Keep only the fields the league manager is meant to see, whatever the page sent.
function cleanSummary_(raw) {
  var s;
  try { s = typeof raw === 'string' ? JSON.parse(raw) : raw; } catch (e) { return null; }
  if (!s || typeof s !== 'object') return null;
  function num(v, max) { v = Number(v); return isNaN(v) ? 0 : Math.max(0, Math.min(max, Math.round(v))); }
  function txt(v, n) { return String(v == null ? '' : v).replace(/[<>]/g, '').slice(0, n); }
  var weak = (Array.isArray(s.weak) ? s.weak : []).slice(0, 3).map(function (w) { return { g: txt(w && w.g, 30), l: txt(w && w.l, 30), p: num(w && w.p, 100) }; });
  return {
    v: 1, week: /^\d{4}-\d{2}-\d{2}$/.test(s.week || '') ? s.week : '', lastActive: /^\d{4}-\d{2}-\d{2}$/.test(s.lastActive || '') ? s.lastActive : '',
    daysThisWeek: num(s.daysThisWeek, 7), gamesThisWeek: num(s.gamesThisWeek, 500), streak: num(s.streak, 5000),
    accuracy: s.accuracy == null ? null : num(s.accuracy, 100), practiceMins: num(s.practiceMins, 10000), weak: weak
  };
}

function statsPush_(b) {
  var a = authPlayer_(b.code, b.deviceId);
  if (a.error) return json_({ ok: false, error: a.error });
  if (!sharesAnywhere_(a.code)) return json_({ ok: false, error: 'not sharing with any league' });
  var clean = cleanSummary_(b.summary);
  if (!clean) return json_({ ok: false, error: 'invalid summary' });
  var text = JSON.stringify(clean);
  if (text.length > LIMITS.maxSummaryChars) return json_({ ok: false, error: 'summary too large' });
  var sh = sheet_(STATS), vals = sh.getDataRange().getValues(), now = new Date().toISOString();
  for (var i = 1; i < vals.length; i++) {
    if (String(vals[i][0]).toUpperCase() === a.code) { sh.getRange(i + 1, 2, 1, 2).setValues([[now, text]]); return json_({ ok: true }); }
  }
  appendText_(sh, 'stats', [a.code, now, text]);
  return json_({ ok: true });
}

// League owner only: the members who chose to share, under First name + initial, with their summary.
function leagueManage_(b) {
  var o = ownerOnly_(b);
  if (o.error) return json_({ ok: false, error: o.error });
  var players = playersByCode_(), roster = membersOf_(memberRows_(), o.lg.id);
  var statRows = sheet_(STATS).getDataRange().getValues(), byCode = {};
  for (var i = 1; i < statRows.length; i++) {
    try { byCode[String(statRows[i][0]).toUpperCase()] = { updatedAt: String(statRows[i][1]), summary: JSON.parse(String(statRows[i][2])) }; } catch (e) { /* skip */ }
  }
  var out = roster.map(function (m) {
    var pl = players[m.code], shared = m.share && m.managerName, st = byCode[m.code];
    var row = { code: m.code, role: m.role, sharing: !!shared, displayName: pl ? pl.displayName : m.code, avatar: pl ? pl.avatar : DEFAULT_AVATAR };
    if (shared) { row.name = m.managerName; row.summary = st ? st.summary : null; row.updatedAt = st ? st.updatedAt : null; }
    else row.reason = m.share ? 'name needed' : 'opted out';
    return row;
  });
  return json_({ ok: true, league: { leagueId: o.lg.id, name: o.lg.name }, members: out, today: today_(), week: mondayOf_(today_()) });
}

// ---------------------------------------------------------------- sync (progress saved online)
// The page makes a recovery code and sends only a hash of it. Saving needs the registered device;
// restoring on a new device needs the friend code plus the recovery code.

function keyRowIndex_(vals, code) {
  for (var i = 1; i < vals.length; i++) if (String(vals[i][0]).toUpperCase() === code && String(vals[i][1]) === 'key') return i;
  return -1;
}
function syncEnable_(b) {
  var a = authPlayer_(b.code, b.deviceId);
  if (a.error) return json_({ ok: false, error: a.error });
  var hash = String(b.keyHash || '');
  if (!/^[a-f0-9]{64}$/.test(hash)) return json_({ ok: false, error: 'invalid key' });
  var sh = sheet_(SYNC), vals = sh.getDataRange().getValues(), idx = keyRowIndex_(vals, a.code), now = new Date().toISOString();
  if (idx >= 0) sh.getRange(idx + 1, 3, 1, 2).setValues([[hash, now]]);
  else appendText_(sh, 'sync', [a.code, 'key', hash, now]);
  return json_({ ok: true });
}
function syncPush_(b) {
  var a = authPlayer_(b.code, b.deviceId);
  if (a.error) return json_({ ok: false, error: a.error });
  var data = String(b.data || '');
  if (!data || data.length > LIMITS.syncChunk * LIMITS.maxSyncChunks) return json_({ ok: false, error: 'saved progress is too large' });
  var sh = sheet_(SYNC), vals = sh.getDataRange().getValues();
  if (keyRowIndex_(vals, a.code) < 0) return json_({ ok: false, error: 'sync is not switched on' });
  for (var i = vals.length - 1; i >= 1; i--) if (String(vals[i][0]).toUpperCase() === a.code && String(vals[i][1]) !== 'key') sh.deleteRow(i + 1);
  var now = new Date().toISOString(), n = Math.ceil(data.length / LIMITS.syncChunk);
  for (var c = 0; c < n; c++) appendText_(sh, 'sync', [a.code, String(c), data.slice(c * LIMITS.syncChunk, (c + 1) * LIMITS.syncChunk), now]);
  return json_({ ok: true, updatedAt: now });
}
function readSnapshot_(code) {
  var vals = sheet_(SYNC).getDataRange().getValues(), parts = [], updatedAt = '';
  for (var i = 1; i < vals.length; i++) {
    if (String(vals[i][0]).toUpperCase() === code && String(vals[i][1]) !== 'key') { parts.push({ n: Number(vals[i][1]), d: String(vals[i][2]) }); updatedAt = String(vals[i][3]); }
  }
  parts.sort(function (x, y) { return x.n - y.n; });
  return { data: parts.map(function (p) { return p.d; }).join(''), updatedAt: updatedAt };
}
function syncPull_(b) {
  var a = authPlayer_(b.code, b.deviceId);
  if (a.error) return json_({ ok: false, error: a.error });
  var snap = readSnapshot_(a.code);
  return json_({ ok: true, data: snap.data || null, updatedAt: snap.updatedAt || null });
}
function syncRestore_(b) {
  var code = cleanCode_(b.code), hash = String(b.keyHash || '');
  if (!code || !/^[a-f0-9]{64}$/.test(hash)) return json_({ ok: false, error: 'check the codes and try again' });
  var cache = CacheService.getScriptCache(), failKey = 'rf:' + code, fails = Number(cache.get(failKey)) || 0;
  if (fails >= LIMITS.maxRestoreTries) return json_({ ok: false, error: 'too many tries — wait an hour and try again' });
  var vals = sheet_(SYNC).getDataRange().getValues(), idx = keyRowIndex_(vals, code);
  if (idx < 0 || String(vals[idx][2]) !== hash) {
    cache.put(failKey, String(fails + 1), 3600);
    return json_({ ok: false, error: 'those codes don’t match' });
  }
  var snap = readSnapshot_(code);
  if (!snap.data) return json_({ ok: false, error: 'nothing has been saved for this code yet' });
  return json_({ ok: true, data: snap.data, updatedAt: snap.updatedAt });
}

// Removes everything stored about a player. Leagues they own must be deleted (or the league left) first.
function deleteMyData_(b) {
  var a = authPlayer_(b.code, b.deviceId);
  if (a.error) return json_({ ok: false, error: a.error });
  var lrows = leagueRows_();
  for (var i = 1; i < lrows.length; i++) if (String(lrows[i][2]).toUpperCase() === a.code) return json_({ ok: false, error: 'delete your leagues first (League → Manage)' });
  deleteRowsWhere_(MEMBERS, 1, a.code);
  deleteRowsWhere_(STATS, 0, a.code);
  deleteRowsWhere_(SYNC, 0, a.code);
  var fr = sheet_(FRIENDS), fv = fr.getDataRange().getValues();
  for (var j = fv.length - 1; j >= 1; j--) if (String(fv[j][0]).toUpperCase() === a.code || String(fv[j][1]).toUpperCase() === a.code) fr.deleteRow(j + 1);
  var sc = sheet_(SCORES), sv = sc.getDataRange().getValues();
  for (var k = sv.length - 1; k >= 1; k--) if (String(sv[k][2]) === String(b.deviceId)) sc.deleteRow(k + 1);
  deleteRowsWhere_(PLAYERS, 0, a.code);
  return json_({ ok: true });
}

// ---------------------------------------------------------------- helpers

function sheet_(name) {
  var ss = SpreadsheetApp.getActiveSpreadsheet(), sh = ss.getSheetByName(name);
  if (!sh) sh = ss.insertSheet(name);
  var h = HEADERS[name];
  if (sh.getLastRow() === 0) sh.getRange(1, 1, 1, h.length).setValues([h]).setFontWeight('bold');
  var props = PropertiesService.getScriptProperties();
  if (props.getProperty('FMT_' + name) !== String(FMT_VERSION)) {
    sh.getRange(1, 1, 1, h.length).setValues([h]).setFontWeight('bold');       // bring the header row up to date when columns are added
    TEXT_COLS[name].forEach(function (c) { sh.getRange(1, c, sh.getMaxRows(), 1).setNumberFormat('@'); });
    props.setProperty('FMT_' + name, String(FMT_VERSION));
  }
  return sh;
}

// appendRow, but guarantee text cells stay text
function appendText_(sh, name, row) {
  var r = sh.getLastRow() + 1;
  TEXT_COLS[name].forEach(function (c) { sh.getRange(r, c).setNumberFormat('@'); });
  sh.getRange(r, 1, 1, row.length).setValues([row]);
}

function playersByCode_() {
  var vals = sheet_(PLAYERS).getDataRange().getValues(), out = {};
  for (var i = 1; i < vals.length; i++) {
    var code = String(vals[i][0]).toUpperCase();
    if (code) out[code] = { code: code, deviceId: String(vals[i][1]), displayName: String(vals[i][2] || code), avatar: String(vals[i][3] || DEFAULT_AVATAR) };
  }
  return out;
}

// Device ids are made by the pages: a UUID, or "id-<time>-<hex>" on very old browsers.
function validDevice_(id) { return typeof id === 'string' && /^[A-Za-z0-9-]{8,64}$/.test(id); }

function countFriends_(vals, code) {

  for (var i = 1; i < vals.length; i++) if (String(vals[i][0]).toUpperCase() === code || String(vals[i][1]).toUpperCase() === code) n++;

// Counts writes in the current minute (called while the script lock is held, so the counts are exact).
function tooManyWrites_(deviceId) {
  var cache = CacheService.getScriptCache(), minute = Math.floor(Date.now() / 60000);
  var keys = ['w:all:' + minute], limits = [LIMITS.writesPerMinuteAll];
  if (deviceId) { keys.push('w:' + String(deviceId).slice(0, 40) + ':' + minute); limits.push(LIMITS.writesPerDevicePerMinute); }
  var counts = cache.getAll(keys), over = false, toSet = {};
  keys.forEach(function (k, i) {
    var n = (Number(counts[k]) || 0) + 1;
    if (n > limits[i]) over = true;
    toSet[k] = String(n);
  });
  cache.putAll(toSet, 120);
  return over;
}

function edgeExists_(vals, a, b) {
  for (var i = 1; i < vals.length; i++) {
    var x = String(vals[i][0]).toUpperCase(), y = String(vals[i][1]).toUpperCase();
    if ((x === a && y === b) || (x === b && y === a)) return true;
  }
  return false;
}

// Sheets may have auto-converted "2026-10-06" to a real date in the past — turn either form back into text.
function normDate_(v) {
  if (v instanceof Date) return Utilities.formatDate(v, SpreadsheetApp.getActiveSpreadsheet().getSpreadsheetTimeZone(), 'yyyy-MM-dd');
  if (typeof v === 'number') return Utilities.formatDate(new Date(Math.round((v - 25569) * 86400000)), 'UTC', 'yyyy-MM-dd');   // a sheet date serial number
  return String(v).slice(0, 10);
}
function today_() { return Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd'); }
function daysBetween_(a, b) {
  var da = new Date(a + 'T00:00:00Z'), db = new Date(b + 'T00:00:00Z');
  return Math.round((da - db) / 86400000);
}

function cleanCode_(c) {
  var s = String(c || '').trim().toUpperCase();
  return /^[A-Z0-9]{4,12}$/.test(s) ? s : '';
}
function cleanName_(n) {
  var t = typeof n === 'string' ? n.replace(/[<>]/g, '').trim().slice(0, 24) : '';
  return t || 'Anonymous';
}
function cleanAvatar_(a) {
  var t = typeof a === 'string' ? a.replace(/[<>]/g, '').trim().slice(0, 16) : '';
  return t || DEFAULT_AVATAR;
}

  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
